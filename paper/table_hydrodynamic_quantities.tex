% Maps the low-order hydrodynamic moments to their spectral basis modes
% Psi_{k,i,l,m}(v,I) = R_{k,l}(|v|) Y_{l,m}(v-hat) H_i(I).
% c denotes the peculiar velocity, I the internal energy, nu the internal-DOF parameter.
\begin{table}[htbp]
  \centering
  \caption{Correspondence between the low-order hydrodynamic moments and the
    spectral basis functions $\Psi_{k,i,l,m}(\mathbf{v},I)
    = R_{k,l}(|\mathbf{v}|)\,Y_{l,m}(\hat{\mathbf{v}})\,H_i(I)$.
    The angular degree $l$ fixes the tensor rank (number of free velocity
    indices) of the moment; the radial/internal degrees $k$/$i$ fix the number
    of energy powers it carries. Here $\mathbf{c}$ is the peculiar velocity and
    $I$ the internal energy.}
  \label{tab:hydrodynamic_quantities}
  \begin{tabular}{@{}llccc@{}}
    \toprule
    Quantity & Characteristic moment & Tensor rank ($l$) & Energy power ($k$/$i$) & Basis mode $(k,i,l)$ \\
    \midrule
    Density / equilibrium               & $1$                                                   & scalar, $l=0$ & $0$                 & $(0,0,0)$ \\
    Momentum                            & $c_i$                                                 & vector, $l=1$ & $0$                 & $(0,0,1)$ \\
    Translational energy / temperature  & $c^2 - \tfrac{3}{2}$                                  & scalar, $l=0$ & $1$ (translational) & $(1,0,0)$ \\
    Internal energy                     & $I - (\nu+1)$                                         & scalar, $l=0$ & $1$ (internal)      & $(0,1,0)$ \\
    Shear stress $\to$ viscosity        & $c_i c_j - \tfrac{1}{3}c^2\delta_{ij}$                & rank-2, $l=2$ & $0$                 & $(0,0,2)$ \\
    Heat flux $\to$ conductivity        & $c_i\!\left(c^2 - \tfrac{5}{2}\right),\ c_i\!\left(I - (\nu+1)\right)$ & vector, $l=1$ & $1$ & $(1,0,1)\ \&\ (0,1,1)$ \\
    \bottomrule
  \end{tabular}
\end{table}
